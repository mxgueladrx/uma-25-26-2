## Introducción
**PL/SQL**: Lenguaje imperativo estructurado cuyo propietario es Oracle. Su estructura es:

```plsql
DECLARE
	nombre_variable tipo; -- opcional
BEGIN
	senetencias pl/sql;
EXCEPTION
	manejo de errores; -- opcional
END;
/
```

## Clases de bloques
- Anónimos.
- Nombrados.
- Procedimientos, paquetes y funciones.
- Disparadores.

## Funciones SQL
- **Caracteres**: CONCAT (||), SUBSTR, LOWER, UPPER, DECODE, LPAD, RPAD...
- **Números**: aritmética, trigonometría, ROUND, TRUNC, GREATEST, LEAST...
- **Fechas**: ADD_MONTHS, MONTHS_BETWEEN, NEXT_DAY, SYSDATE...
- **Funciones de conversión**: TO_NUMBER, TO_CHAR, TO_DATE
- **Pseudo-columnas**: ROWNUM, ROWID
- **Manejo de valores nulos**: NVL

## Secuencias
Elementos productores de valores numéricos independientes del resto de objetos del esquema.

```plsql
CREATE SEQUENCE nombre START WITH primer_valor INCREMENT BY numero;

nombre.NEXTVAL; -- siguiente valor
nombre.CURRVAL; -- valor actual
```

## Identificadores
Cadenas de caracteres que comienzan por una letra y longitud menor que 30.

## Variables

```plsql
nombre [CONSTANT] datatype [NOT NULL] [:= expr | DEFAULT expr];
```

**Ámbito de la variable**: desde el BEGIN al END. 
**Visibilidad**: son visibles en su ámbito salvo sobrecarga con otra definición posterior. Uso de bloques nombrados para aumentar visibilidad.

**Registros**: Se puede asignar componente a componente o registros enteros, siempre que sean del mismo tipo.

```plsql
TYPE nombre IS RECORD ( 
	campo_1 tip_1 [NOT NULL] [:= valor 1],
	... , 
	campo_k tipo_k [NOT NULL] [:= valor k] 
);
```

- `%TYPE`: Copia el tipo de dato de otra variable existente o columna de una tabla.
- `%ROWTYPE`: Declara una variable de tipo registro que contiene los campos de todas las columnas de una tabla.

## Control de flujo
- **Condicionales**: 

```plsql
-- if-else
IF expresion_1 THEN 
	cuerpo_1; 
ELSIF expresión_2 THEN 
	cuerpo_2;
ELSE 
	cuerpo_3; 
END IF;

-- case-when
CASE expresion_1 
WHEN valor_1 THEN 
	cuerpo_1;
WHEN valor_2 THEN 
	cuerpo_2; 
ELSE 
	cuerpo_n;
END CASE;
```

- **Bucles**:

```plsql
-- loop
etiqueta LOOP
	cuerpo;
	EXIT etiqueta WHEN condicion;
END LOOP etiqueta;

-- while
etiqueta WHILE condicion LOOP
	cuerpo;
END LOOP etiqueta;

-- for
etiqueta FOR contador IN [REVERSE] valor_inicio..valor_fin LOOP
	cuerpo;
END LOOP etiqueta;
```

## Manipulación de datos
- Se puede meter DML en PL/SQL.
- Se pueden usar variables o literales (que no sean nombre de tablas o columnas).
- La salida de un `SELECT` se puede volcar a una variable:

```plsql
SELECT ... INTO variables | registro FROM ... WHERE ...;
```

## Cursores
**Puntero**: Área de trabajo de memoria privada (context area) dedicada a la ejecución de sentencias.

```plsql
CURSOR nombre IS SELECT ... FROM ... WHERE ...;

-- abrir y cerrar cursor
OPEN nombre_cursor;
CLOSE nombre_cursor;

-- recorrido de cursor con for
DECLARE 
	CURSOR cur_mensajes IS SELECT codigo, texto FROM mensajes; 
BEGIN 
	FOR v_mensaje IN cur_mensajes LOOP
		DBMS_OUTPUT.PUT_LINE(v_mensaje.codigo||' '||v_mensaje.texto);
	END LOOP;
END;

-- con parametros para filtrar resultados de consulta
DECLARE 
	CURSOR cur_mensajes (p_patron VARCHAR2) IS SELECT codigo, texto FROM mensajes where texto like '%'||p_patron||'%'; 
	v_patron varchar2(100); 
BEGIN 
	v_patron := 'Hola'; 
	FOR v_mensaje IN cur_mensajes(v_patron) LOOP
		DBMS_OUTPUT.PUT_LINE(v_mensaje.codigo||' '||v_mensaje.texto); 
	END LOOP;
END;

-- sin declarar cursor
BEGIN 
	FOR v_mensaje IN (SELECT codigo, texto FROM mensajes) LOOP
		DBMS_OUTPUT.PUT_LINE(v_mensaje.codigo||' '||v_mensaje.texto); 
	END LOOP;
END;

-- con actualizacion para modificar filas
DECLARE 
	CURSOR c_cursor IS SELECT... FROM tabla WHERE... FOR UPDATE; 
BEGIN
	FOR var_cursor IN c_cursor LOOP 
		UPDATE tabla SET ... WHERE CURRENT OF c_cursor; 
	END LOOP; 
	COMMIT; 
END;
```

## SQL dinámico
Lanzar sentencias que no están definidas anteriormente. `EXECUTE IMMEDIATE ...`

```plsql
DECLARE 
	var_tabla VARCHAR2(100); 
BEGIN 
	var_tabla:='EMPLEADOS'; 
	EXECUTE IMMEDIATE 'INSERT INTO '||var_tabla||' VALUES (1,''Yo'')'; 
	COMMIT; 
END;
```

## Procedimientos
Bloques PL/SQL con nombre, compilados y almacenados en el servidor.

```plsql
CREATE [OR REPLACE] PROCEDURE nombre_procedimiento [(parametro_1 [{IN|OUT|IN OUT}] tipo_1, ..., parametro_n [{IN|OUT|IN OUT}] tipo_n)] {IS|AS} 
--declaracion de variables opcional 
BEGIN 
	... -- tratamiento con PL/SQL 
[EXCEPTION ...] -- opcional 
END [nombre_procedimiento];

EXECUTE nombre_procedimiento(parametros)

DROP PROCEDURE nombre_procedimiento
```

## Funciones

```plsql
CREATE [OR REPLACE] FUNCTION nombre_funcion [(parametros)] RETURN tipo_valor_devuelto {IS|AS} 
--declaracion de variables opcional 
BEGIN 
	... -- Cuerpo Funcion 
RETURN valor_devuelto; 
END nombre_funcion;

DROP FUNCTION nombre_funcion
```

## Paquetes
Asociación de un conjunto de objetos (procedimientos, funciones, variables, excepciones, tipos y cursores) y se pueden agrupar según varios criterios.

```plsql
-- declaracion de especificacion
CREATE OR REPLACE PACKAGE nombre_paquete AS 
	PROCEDURE procedimiento(parametro tipo);
	FUNCTION funcion(arg1 tipo) RETURN tipo_devuelto; 
	... --especificacion de objetos
END nombre_paquete;

-- declaracion del cuerpo
CREATE OR REPLACE PACKAGE BODY nombre_paquete AS 
	PROCEDURE procedimiento(parametro tipo) IS 
	BEGIN 
		--cuerpo del procedimiento 
	END; 
	
	FUNCTION funcion(arg1 tipo) RETURN tipo_devuelto IS 
	BEGIN 
		--cuerpo de la funcion 
	END;
	
	...
END nombre_paquete;
```

## Excepciones

```plsql
DECLARE
	exc_error EXCEPTION;
BEGIN
	IF condicion THEN
		RAISE exc_error;
	END IF;
EXCEPTION
	WHEN exc_error THEN ... -- capturar error especifico
	WHEN OTHERS THEN ... -- cualquier otro error
END;
```
